rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Public read for all documents in the 'designs' collection
    match /designs/{designId} {
      allow read: if true;
      // Only authenticated users can create (publish) a new design
      allow create: if request.auth != null;
      // Only the original author (matching the document's authorId field) can update or delete
      allow update, delete: if request.auth != null && request.resource.data.author.uid == request.auth.uid;
    }
    
    // Comments collection
    match /comments/{commentId} {
      allow read: if true;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null && request.resource.data.author.uid == request.auth.uid;
    }
    
    // User profiles
    match /profiles/{userId} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId;
    }
  }
}
